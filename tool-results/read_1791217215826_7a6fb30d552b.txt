   176→/* ================= CHECKOUT ================= */
   177→
   178→export function CheckoutDialog() {
   179→  const open = useMarketplace((s) => s.checkoutOpen);
   180→  const setOpen = useMarketplace((s) => s.setCheckoutOpen);
   181→  const cart = useMarketplace((s) => s.cart);
   182→  const clearCart = useMarketplace((s) => s.clearCart);
   183→  const setSuccessOrder = useMarketplace((s) => s.setSuccessOrder);
   184→  const lastEmail = useMarketplace((s) => s.lastEmail);
   185→  const setLastEmail = useMarketplace((s) => s.setLastEmail);
   186→  const { toast } = useToast();
   187→  const { t } = useI18n();
   188→
   189→  const [email, setEmail] = useState("");
   190→  const [payment, setPayment] = useState<string>("card");
   191→  const [submitting, setSubmitting] = useState(false);
   192→  const [error, setError] = useState<string | null>(null);
   193→
   194→  useEffect(() => {
   195→    if (open) {
   196→      setEmail(lastEmail);
   197→      setError(null);
   198→    }
   199→  }, [open, lastEmail]);
   200→
   201→  const subtotal = cartSubtotal(cart);
   202→  const fee = serviceFee(subtotal);
   203→
   204→  const pay = async () => {
   205→    if (submitting) return;
   206→    setSubmitting(true);
   207→    setError(null);
   208→    try {
   209→      const res = await fetch("/api/checkout", {
   210→        method: "POST",
   211→        headers: { "Content-Type": "application/json" },
   212→        body: JSON.stringify({
   213→          email,
   214→          paymentMethod: payment,
   215→          items: cart.map((l) => ({ slug: l.slug, quantity: l.quantity })),
   216→        }),
   217→      });
   218→      const data = await res.json();
   219→      if (!res.ok) throw new Error(data.error ?? "Payment failed");
   220→      setOpen(false);
   221→      setSuccessOrder(data.order as ApiOrder);
   222→      setLastEmail(email);
   223→      clearCart();
   224→    } catch (e) {
   225→      const msg = e instanceof Error ? e.message : "Payment failed";
   226→      setError(msg);
   227→      toast({ title: t("checkout.failed"), description: msg, variant: "destructive" });
   228→    } finally {
   229→      setSubmitting(false);
   230→    }
   231→  };
   232→
   233→  return (
   234→    <Dialog open={open} onOpenChange={setOpen}>
   235→      <DialogContent className="max-h-[92vh] overflow-y-auto border-white/10 bg-[oklch(0.16_0.02_305)] sm:max-w-lg">
   236→        <DialogHeader>
   237→          <DialogTitle className="flex items-center gap-2 text-xl text-white">
   238→            <Zap className="h-5 w-5 text-amber-300" />
   239→            {t("checkout.title")}
   240→          </DialogTitle>
   241→          <DialogDescription>{t("checkout.desc")}</DialogDescription>
   242→        </DialogHeader>
   243→
   244→        <div className="space-y-4">
   245→          <div className="grid gap-1.5">
   246→            <Label htmlFor="buyer-email" className="text-sm text-foreground/90">
   247→              {t("checkout.email")}
   248→            </Label>
   249→            <Input
   250→              id="buyer-email"
   251→              type="email"
   252→              value={email}
   253→              onChange={(e) => setEmail(e.target.value)}
   254→              placeholder="you@example.com"
   255→              className="h-11 border-white/10 bg-white/5"
   256→              dir="ltr"
   257→            />
   258→          </div>
   259→
   260→          <div className="grid gap-1.5">
   261→            <Label className="text-sm text-foreground/90">{t("checkout.method")}</Label>
   262→            <RadioGroup value={payment} onValueChange={setPayment} className="grid grid-cols-2 gap-2">
   263→              {PAYMENT_IDS.map((id) => (
   264→                <Label
   265→                  key={id}
   266→                  htmlFor={`pay-${id}`}
   267→                  className={`flex cursor-pointer items-center gap-3 rounded-2xl border p-3.5 transition ${
   268→                    payment === id

   274→                  <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-white/5 text-fuchsia-200">
   275→                    {id === "card" ? (
   276→                      <CreditCard className="h-5 w-5" />
   277→                    ) : id === "paypal" ? (
   278→                      <Wallet className="h-5 w-5" />

   286→                    <span className="block text-[13px] font-bold text-foreground">

   492→/* ================= MY ORDERS ================= */
   493→
   494→export function OrdersDialog() {
   495→  const open = useMarketplace((s) => s.ordersOpen);
   496→  const setOpen = useMarketplace((s) => s.setOrdersOpen);
   497→  const lastEmail = useMarketplace((s) => s.lastEmail);
   498→  const { toast } = useToast();
   499→  const { t } = useI18n();
   500→
   501→  const [email, setEmail] = useState("");
   502→  const [orders, setOrders] = useState<ApiOrder[] | null>(null);
   503→  const [loading, setLoading] = useState(false);
   504→
   505→  useEffect(() => {
   506→    if (open) {
   507→      setEmail(lastEmail);
   508→      setOrders(null);
   509→    }
   510→  }, [open, lastEmail]);
   511→
   512→  const lookup = async () => {
   513→    if (loading || !email.trim()) return;
   514→    setLoading(true);
   515→    try {
   516→      const res = await fetch(`/api/orders?email=${encodeURIComponent(email.trim())}`);
   517→      const data = await res.json();
   518→      if (!res.ok) throw new Error(data.error ?? "Lookup failed");
   519→      setOrders(data.orders ?? []);
   520→      if ((data.orders ?? []).length === 0) {
   521→        toast({
   522→          title: t("orders.noneTitle"),
   523→          description: t("orders.noneText", { email: email.trim() }),
   524→        });
   525→      }
   526→    } catch (e) {
   527→      toast({
   528→        title: t("orders.failed"),
