   177→   177→
   178→   178→export function CheckoutDialog() {
   179→   179→  const open = useMarketplace((s) => s.checkoutOpen);
   180→   180→  const setOpen = useMarketplace((s) => s.setCheckoutOpen);
   181→   181→  const cart = useMarketplace((s) => s.cart);
   182→   182→  const clearCart = useMarketplace((s) => s.clearCart);
   183→   183→  const setSuccessOrder = useMarketplace((s) => s.setSuccessOrder);
   184→   184→  const lastEmail = useMarketplace((s) => s.lastEmail);
   185→   185→  const setLastEmail = useMarketplace((s) => s.setLastEmail);
   186→   186→  const { toast } = useToast();
   187→   187→  const { t } = useI18n();
   188→   188→
   189→   189→  const [email, setEmail] = useState("");
   190→   190→  const [payment, setPayment] = useState<string>("card");
   191→   191→  const [submitting, setSubmitting] = useState(false);
   192→   192→  const [error, setError] = useState<string | null>(null);
   193→   193→
   194→   194→  useEffect(() => {
   195→   195→    if (open) {
   196→   196→      setEmail(lastEmail);
   197→   197→      setError(null);
   198→   198→    }
   199→   199→  }, [open, lastEmail]);
   200→   200→
   201→   201→  const subtotal = cartSubtotal(cart);
   202→   202→  const fee = serviceFee(subtotal);
   203→   203→
   204→   204→  const pay = async () => {
   205→   205→    if (submitting) return;
   206→   206→    setSubmitting(true);
   207→   207→    setError(null);
   208→   208→    try {
   209→   209→      const res = await fetch("/api/checkout", {
   210→   210→        method: "POST",
   211→   211→        headers: { "Content-Type": "application/json" },
   212→   212→        body: JSON.stringify({
   213→   213→          email,
   214→   214→          paymentMethod: payment,
   215→   215→          items: cart.map((l) => ({ slug: l.slug, quantity: l.quantity })),
   216→   216→        }),
   217→   217→      });
   218→   218→      const data = await res.json();
   219→   219→      if (!res.ok) throw new Error(data.error ?? "Payment failed");
   220→   220→      setOpen(false);
   221→   221→      setSuccessOrder(data.order as ApiOrder);
   222→   222→      setLastEmail(email);
   223→   223→      clearCart();
   224→   224→    } catch (e) {
   225→   225→      const msg = e instanceof Error ? e.message : "Payment failed";
   226→   226→      setError(msg);
   227→   227→      toast({ title: t("checkout.failed"), description: msg, variant: "destructive" });
   228→   228→    } finally {
   229→   229→      setSubmitting(false);
   230→   230→    }
   231→   231→  };
   232→   232→
   233→   233→  return (
   234→   234→    <Dialog open={open} onOpenChange={setOpen}>
   235→   235→      <DialogContent className="max-h-[92vh] overflow-y-auto border-white/10 bg-[oklch(0.16_0.02_305)] sm:max-w-lg">
   236→   236→        <DialogHeader>
   237→   237→          <DialogTitle className="flex items-center gap-2 text-xl text-white">
   238→   238→            <Zap className="h-5 w-5 text-amber-300" />
   239→   239→            {t("checkout.title")}
   240→   240→          </DialogTitle>
   241→   241→          <DialogDescription>{t("checkout.desc")}</DialogDescription>
   242→   242→        </DialogHeader>
   243→   243→
   244→   244→        <div className="space-y-4">
   245→   245→          <div className="grid gap-1.5">
   246→   246→            <Label htmlFor="buyer-email" className="text-sm text-foreground/90">
   247→   247→              {t("checkout.email")}
   248→   248→            </Label>
   249→   249→            <Input
   250→   250→              id="buyer-email"
   251→   251→              type="email"
   252→   252→              value={email}
   253→   253→              onChange={(e) => setEmail(e.target.value)}
   254→   254→              placeholder="you@example.com"
   255→   255→              className="h-11 border-white/10 bg-white/5"
   256→   256→              dir="ltr"
   257→   257→            />
   258→   258→          </div>
   259→   259→
   260→   260→          <div className="grid gap-1.5">
   261→   261→            <Label className="text-sm text-foreground/90">{t("checkout.method")}</Label>
   262→   262→            <RadioGroup value={payment} onValueChange={setPayment} className="grid grid-cols-2 gap-2">
   263→   263→              {PAYMENT_IDS.map((id) => (
   264→   264→                <Label
   265→   265→                  key={id}
   266→   266→                  htmlFor={`pay-${id}`}
   267→   267→                  className={`flex cursor-pointer items-center gap-3 rounded-2xl border p-3.5 transition ${
   268→   268→                    payment === id
   269→   269→                      ? "border-fuchsia-400/60 bg-fuchsia-500/10"
   270→   270→                      : "border-white/10 bg-white/[0.03] hover:border-white/25"
   271→   271→                  }`}
   272→   272→                >
   273→   273→                  <RadioGroupItem id={`pay-${id}`} value={id} className="sr-only" />
   274→   274→                  <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-white/5 text-fuchsia-200">
   275→   275→                    {id === "card" ? (
   276→   276→                      <CreditCard className="h-5 w-5" />
   277→   277→                    ) : id === "paypal" ? (
   278→   278→                      <Wallet className="h-5 w-5" />
   279→   279→                    ) : id === "crypto" ? (
   280→   280→                      <Bitcoin className="h-5 w-5" />
   281→   281→                    ) : (
   282→   282→                      <Smartphone className="h-5 w-5" />
   283→   283→                    )}
   284→   284→                  </span>
   285→   285→                  <span>
   286→   286→                    <span className="block text-[13px] font-bold text-foreground">
   287→   287→                      {t(`checkout.${id}`)}
   288→   288→                    </span>
   289→   289→                    <span className="block text-[11px] text-muted-foreground">
   290→   290→                      {t(`checkout.${id}Hint`)}
   291→   291→                    </span>
   292→   292→                  </span>
   293→   293→                </Label>
   294→   294→              ))}
   295→   295→            </RadioGroup>
   296→   296→          </div>
   297→   297→
   298→   298→          <div className="rounded-2xl border border-white/10 bg-white/[0.03] p-4">
   299→   299→            {cart.map((l) => (
   300→   300→              <div
   301→   301→                key={l.slug}
   302→   302→                className="flex items-center justify-between gap-2 py-1 text-sm"
   303→   303→              >
   304→   304→                <span className="min-w-0 flex-1 truncate text-foreground/85">
   305→   305→                  {l.emoji} {l.title} <span className="text-muted-foreground">×{l.quantity}</span>
   306→   306→                </span>
   307→   307→                <span className="font-semibold text-foreground/85">
   308→   308→                  {formatMoney(l.price * l.quantity)}
   309→   309→                </span>
   310→   310→              </div>
   311→   311→            ))}
   312→   312→            <Separator className="my-2 bg-white/10" />
   313→   313→            <div className="flex justify-between py-0.5 text-xs text-muted-foreground">
   314→   314→              <span>{t("checkout.serviceFee")}</span>
   315→   315→              <span>{formatMoney(fee)}</span>
   316→   316→            </div>
   317→   317→            <div className="flex justify-between pt-1 text-base font-bold text-white">
   318→   318→              <span>{t("cart.total")}</span>
   319→   319→              <span>{formatMoney(subtotal + fee)}</span>
   320→   320→            </div>
   321→   321→          </div>
   322→   322→
   323→   323→          {error ? (
   324→   324→            <p className="rounded-xl border border-rose-400/30 bg-rose-500/10 px-3.5 py-2.5 text-sm text-rose-200">
   325→   325→              {error}
   326→   326→            </p>
   327→   327→          ) : null}
   328→   328→
   329→   329→          <Button
   330→   330→            onClick={pay}
   331→   331→            disabled={submitting || cart.length === 0}
   332→   332→            className="h-12 w-full rounded-2xl bg-gradient-to-r from-violet-500 to-fuchsia-500 text-[15px] font-bold text-white shadow-lg shadow-fuchsia-500/25 hover:from-violet-400 hover:to-fuchsia-400"
   333→   333→          >
   334→   334→            {submitting ? (
