   184→   184→
   185→   185→export function CheckoutDialog() {
   186→   186→  const open = useMarketplace((s) => s.checkoutOpen);
   187→   187→  const setOpen = useMarketplace((s) => s.setCheckoutOpen);
   188→   188→  const cart = useMarketplace((s) => s.cart);
   189→   189→  const clearCart = useMarketplace((s) => s.clearCart);
   190→   190→  const setSuccessOrder = useMarketplace((s) => s.setSuccessOrder);
   191→   191→  const lastEmail = useMarketplace((s) => s.lastEmail);
   192→   192→  const setLastEmail = useMarketplace((s) => s.setLastEmail);
   193→   193→  const { toast } = useToast();
   194→   194→
   195→   195→  const [email, setEmail] = useState("");
   196→   196→  const [payment, setPayment] = useState("card");
   197→   197→  const [submitting, setSubmitting] = useState(false);
   198→   198→  const [error, setError] = useState<string | null>(null);
   199→   199→
   200→   200→  useEffect(() => {
   201→   201→    if (open) {
   202→   202→      setEmail(lastEmail);
   203→   203→      setError(null);
   204→   204→    }
   205→   205→  }, [open, lastEmail]);
   206→   206→
   207→   207→  const subtotal = cartSubtotal(cart);
   208→   208→  const fee = serviceFee(subtotal);
   209→   209→
   210→   210→  const pay = async () => {
   211→   211→    if (submitting) return;
   212→   212→    setSubmitting(true);
   213→   213→    setError(null);
   214→   214→    try {
   215→   215→      const res = await fetch("/api/checkout", {
   216→   216→        method: "POST",
   217→   217→        headers: { "Content-Type": "application/json" },
   218→   218→        body: JSON.stringify({
   219→   219→          email,
   220→   220→          paymentMethod: payment,
   221→   221→          items: cart.map((l) => ({ slug: l.slug, quantity: l.quantity })),
   222→   222→        }),
   223→   223→      });
   224→   224→      const data = await res.json();
   225→   225→      if (!res.ok) throw new Error(data.error ?? "Payment failed");
   226→   226→      setOpen(false);
   227→   227→      setSuccessOrder(data.order as ApiOrder);
   228→   228→      setLastEmail(email);
   229→   229→      clearCart();
   230→   230→    } catch (e) {
   231→   231→      const msg = e instanceof Error ? e.message : "Payment failed";
   232→   232→      setError(msg);
   233→   233→      toast({ title: "Payment failed", description: msg, variant: "destructive" });
   234→   234→    } finally {
   235→   235→      setSubmitting(false);
   236→   236→    }
   237→   237→  };
   238→   238→
   239→   239→  return (
   240→   240→    <Dialog open={open} onOpenChange={setOpen}>
   241→   241→      <DialogContent className="max-h-[92vh] overflow-y-auto border-white/10 bg-[oklch(0.16_0.02_305)] sm:max-w-lg">
   242→   242→        <DialogHeader>
   243→   243→          <DialogTitle className="flex items-center gap-2 text-xl text-white">
   244→   244→            <Zap className="h-5 w-5 text-amber-300" />
   245→   245→            Secure checkout
   246→   246→          </DialogTitle>
   247→   247→          <DialogDescription>
   248→   248→            Goods are delivered to your email + on screen immediately after payment.
   249→   249→          </DialogDescription>
   250→   250→        </DialogHeader>
   251→   251→
   252→   252→        <div className="space-y-4">
   253→   253→          <div className="grid gap-1.5">
   254→   254→            <Label htmlFor="buyer-email" className="text-sm text-foreground/90">
   255→   255→              Delivery email
   256→   256→            </Label>
   257→   257→            <Input
   258→   258→              id="buyer-email"
   259→   259→              type="email"
   260→   260→              value={email}
   261→   261→              onChange={(e) => setEmail(e.target.value)}
   262→   262→              placeholder="you@example.com"
   263→   263→              className="h-11 border-white/10 bg-white/5"
   264→   264→            />
   265→   265→          </div>
   266→   266→
   267→   267→          <div className="grid gap-1.5">
   268→   268→            <Label className="text-sm text-foreground/90">Payment method</Label>
   269→   269→            <RadioGroup
   270→   270→              value={payment}
   271→   271→              onValueChange={setPayment}

   273→   273→            >
   274→   274→              {PAYMENTS.map((p) => (
   275→   275→                <Label
   276→   276→                  key={p.id}
   277→   277→                  htmlFor={`pay-${p.id}`}
   278→   278→                  className={`flex cursor-pointer items-center gap-3 rounded-2xl border p-3.5 transition ${
   279→   279→                    payment === p.id
   280→   280→                      ? "border-fuchsia-400/60 bg-fuchsia-500/10"
