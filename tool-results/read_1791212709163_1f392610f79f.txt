   112→
   113→export async function adminFetch(url: string, init?: RequestInit): Promise<Response> {
   114→  const token = typeof window !== "undefined" ? localStorage.getItem(TOKEN_KEY) ?? "" : "";
   115→  const res = await fetch(url, {
   116→    ...init,
   117→    headers: {
   118→      "Content-Type": "application/json",
   119→      "x-admin-key": token,
   120→      ...(init?.headers ?? {}),
   121→    },
   122→  });
   123→  return res;
   124→}

   231→              <ShieldCheck className="h-7 w-7 text-white" />

   405→            await loadCatalog();
   406→          }}
   407→        />
   408→      ) : null}
   409→    </div>
   410→  );

   437→function ProductsTab({
   438→  data,
   439→  loading,
   440→  onEdit,
   441→  onDeleted,
   442→  onOpenAdd,
   443→}: {
   444→  data: AdminData | null;
   445→  loading: boolean;
   446→  onEdit: (p: AdminProduct) => void;
   447→  onDeleted: () => Promise<void>;
   448→  onOpenAdd: () => void;
   449→}) {
   450→  const { toast } = useToast();
   451→  const { t, cat } = useI18n();
   452→  const [query, setQuery] = useState("");
   453→  const [deleting, setDeleting] = useState<string | null>(null);
   454→
   455→  const products = useMemo(() => {
   456→    const list = data?.products ?? [];
   457→    const q = query.trim().toLowerCase();
   458→    if (!q) return list;
   459→    return list.filter((p) => `${p.title} ${p.brand} ${p.categoryName}`.toLowerCase().includes(q));
   460→  }, [data, query]);
   461→
   462→  const toggleActive = async (p: AdminProduct) => {
   463→    try {
   464→      const res = await adminFetch(`/api/admin/products/${p.id}`, {
   465→        method: "PATCH",
   466→        body: JSON.stringify({ isActive: !p.isActive }),
   467→      });
   468→      if (!res.ok) throw new Error();
   469→      await onDeleted();
   470→    } catch {
   471→      toast({ title: "Update failed", variant: "destructive" });
   472→    }
   473→  };
   474→
   475→  const remove = async (p: AdminProduct) => {
   476→    if (deleting) return;
   477→    if (!window.confirm(t("admin.deleteConfirm"))) return;
   478→    setDeleting(p.id);
   479→    try {
   480→      const res = await adminFetch(`/api/admin/products/${p.id}`, { method: "DELETE" });
   481→      if (!res.ok) throw new Error();
   482→      toast({ title: t("admin.productDeleted"), description: p.title });
   483→      await onDeleted();
   484→    } catch {
   485→      toast({ title: "Delete failed", variant: "destructive" });
   486→    } finally {
   487→      setDeleting(null);
   488→    }
   489→  };
   490→
   491→  return (
   492→    <section className="mt-4">
   493→      <div className="mb-3 flex flex-wrap items-center gap-2">
   494→        <div className="relative min-w-0 flex-1">
   495→          <Search className="pointer-events-none absolute start-3.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   496→          <Input
   497→            value={query}
   498→            onChange={(e) => setQuery(e.target.value)}
   499→            placeholder={t("admin.searchPlaceholder")}
   500→            className="h-10 rounded-xl border-white/10 bg-white/5 ps-10"
   501→          />
   502→        </div>
   503→        <Button
   504→          onClick={onOpenAdd}
   505→          className="h-10 gap-2 rounded-xl bg-gradient-to-r from-emerald-500 to-teal-500 font-bold text-white shadow-lg shadow-emerald-500/20 sm:hidden"
   506→        >
   507→          <Plus className="h-4 w-4" />
   508→          {t("admin.addProduct")}
   509→        </Button>
   510→      </div>
   511→
   512→      {loading ? (
   513→        <div className="flex items-center justify-center rounded-2xl border border-white/10 bg-card/50 py-16">
   514→          <Loader2 className="h-6 w-6 animate-spin text-fuchsia-300" />
   515→        </div>
   516→      ) : products.length === 0 ? (
   517→        <div className="rounded-2xl border border-dashed border-white/15 py-16 text-center">
   518→          <p className="text-3xl">📦</p>
   519→          <p className="mt-2 text-sm text-muted-foreground">{t("admin.noProducts")}</p>
   520→        </div>
   521→      ) : (
   522→        <div className="nice-scroll max-h-[62vh] overflow-auto rounded-2xl border border-white/10">
   523→          <table className="w-full min-w-[860px] text-sm">

   978→type CodeRow = {
   979→  id: string;
   980→  value: string;
   981→  sold: boolean;
   982→  orderShortId: string | null;
   983→  createdAt: string;
   984→};
   985→
   986→function CodesTab({ data, onChanged }: { data: AdminData | null; onChanged: () => Promise<void> }) {
   987→  const { toast } = useToast();
   988→  const { t } = useI18n();
   989→
   990→  const products = data?.products ?? [];
   991→  const [productId, setProductId] = useState<string>("");
   992→  const [codes, setCodes] = useState<CodeRow[]>([]);
   993→  const [loadingCodes, setLoadingCodes] = useState(false);
   994→  const [bulk, setBulk] = useState("");
   995→  const [adding, setAdding] = useState(false);
   996→  const [deletingId, setDeletingId] = useState<string | null>(null);
   997→
   998→  const selected = products.find((p) => p.id === productId) ?? null;
   999→
  1000→  useEffect(() => {
  1001→    if (!productId && products.length > 0) {
  1002→      setProductId(products[0].id);
  1003→    }
  1004→  }, [products, productId]);
  1005→
  1006→  useEffect(() => {
  1007→    if (!productId) return;
  1008→    setLoadingCodes(true);
  1009→    setCodes([]);
  1010→    adminFetch(`/api/admin/products/${productId}/codes`)
  1011→      .then((r) => r.json())
  1012→      .then((d) => setCodes(d.codes ?? []))
  1013→      .catch(() => setCodes([]))
  1014→      .finally(() => setLoadingCodes(false));
  1015→  }, [productId, data]);
  1016→
  1017→  const addCodes = async () => {
  1018→    if (adding || !productId || !bulk.trim()) return;
  1019→    setAdding(true);
  1020→    try {
  1021→      const res = await adminFetch(`/api/admin/products/${productId}/codes`, {
  1022→        method: "POST",
  1023→        body: JSON.stringify({ codes: bulk }),
  1024→      });
  1025→      const d = await res.json();
  1026→      if (!res.ok) throw new Error(d.error ?? "Could not add codes");
  1027→      toast({
  1028→        title: t("admin.codesAdded", { n: d.added ?? 0 }),
  1029→        description: d.skipped > 0 ? t("admin.codesDuplicate", { n: d.skipped }) : undefined,
  1030→      });
  1031→      setBulk("");
  1032→      const listRes = await adminFetch(`/api/admin/products/${productId}/codes`);
  1033→      const listData = await listRes.json();
  1034→      setCodes(listData.codes ?? []);
  1035→      await onChanged();
  1036→    } catch (e) {
  1037→      toast({
  1038→        title: "Failed to add codes",
  1039→        description: e instanceof Error ? e.message : "",
  1040→        variant: "destructive",
  1041→      });
  1042→    } finally {
  1043→      setAdding(false);
  1044→    }
  1045→  };
  1046→
  1047→  const deleteCode = async (code: CodeRow) => {
  1048→    if (deletingId) return;
  1049→    setDeletingId(code.id);
  1050→    try {
  1051→      const res = await adminFetch(`/api/admin/codes/${code.id}`, { method: "DELETE" });
  1052→      if (!res.ok) {
  1053→        const d = await res.json();
  1054→        throw new Error(d.error ?? "Delete failed");
  1055→      }
  1056→      setCodes((prev) => prev.filter((c) => c.id !== code.id));
  1057→      await onChanged();
